cmake_minimum_required(VERSION 3.16)
set(PROJECT_VER "0.1.0-lab")
if(EOTA_LAB_ARMED)
    if(NOT EOTA_LAB_INPUTS OR NOT EOTA_LAB_SDKCONFIG)
        message(FATAL_ERROR "Armed ESP32 build requires external lab inputs and sdkconfig defaults")
    endif()
    foreach(_input IN ITEMS EOTA_LAB_INPUTS EOTA_LAB_SDKCONFIG)
        if(NOT IS_ABSOLUTE "${${_input}}" OR NOT EXISTS "${${_input}}")
            message(FATAL_ERROR "${_input} must name an existing absolute path")
        endif()
        get_filename_component(_input_real "${${_input}}" REALPATH)
        get_filename_component(_source_real "${CMAKE_CURRENT_LIST_DIR}/../.." REALPATH)
        get_filename_component(_build_real "${CMAKE_BINARY_DIR}" REALPATH)
        string(FIND "${_input_real}" "${_source_real}/" _source_prefix)
        string(FIND "${_input_real}" "${_build_real}/" _build_prefix)
        if(_source_prefix EQUAL 0 OR _build_prefix EQUAL 0)
            message(FATAL_ERROR "${_input} must be outside this repository and build directory")
        endif()
    endforeach()
    set(SDKCONFIG_DEFAULTS "sdkconfig.defaults;${EOTA_LAB_SDKCONFIG}")
endif()
set(EXTRA_COMPONENT_DIRS "${CMAKE_CURRENT_LIST_DIR}/../../components/esp_ota")
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(esp_ota_lab)
